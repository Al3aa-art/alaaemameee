<script>
const questions = [
 {word:"كتاب", answer:"اسم"},
 {word:"يكتب", answer:"فعل"},
 {word:"من", answer:"حرف"},
 {word:"مدرسة", answer:"اسم"},
 {word:"ذهب", answer:"فعل"},
 {word:"إلى", answer:"حرف"},
 {word:"قلم", answer:"اسم"},
 {word:"يلعب", answer:"فعل"},
 {word:"على", answer:"حرف"},
 {word:"طالب", answer:"اسم"},
 {word:"قرأ", answer:"فعل"},
 {word:"في", answer:"حرف"},
 {word:"شجرة", answer:"اسم"},
 {word:"نام", answer:"فعل"},
 {word:"عن", answer:"حرف"},
 {word:"طاولة", answer:"اسم"},
 {word:"كتب", answer:"فعل"},
 {word:"حتى", answer:"حرف"},
 {word:"باب", answer:"اسم"},
 {word:"شرب", answer:"فعل"},
 {word:"ثم", answer:"حرف"},
 {word:"قمر", answer:"اسم"},
 {word:"جلس", answer:"فعل"},
 {word:"أو", answer:"حرف"},
 {word:"سيارة", answer:"اسم"},
 {word:"ركب", answer:"فعل"},
 {word:"بل", answer:"حرف"},
 {word:"ولد", answer:"اسم"},
 {word:"ضحك", answer:"فعل"},
 {word:"لكن", answer:"حرف"}
];

let index = 0;
let score = 0;
let time = 30;
let timer;

function loadQuestion() {
    if (index >= questions.length) {
        document.body.innerHTML = `<h2>🎉 انتهت اللعبة</h2><p>درجتك النهائية: ${score} / 30</p>`;
        return;
    }

    document.getElementById("question").innerText =
        `ما نوع الكلمة: (${questions[index].word}) ؟`;

    const choicesDiv = document.getElementById("choices");
    choicesDiv.innerHTML = "";

    ["اسم","فعل","حرف"].forEach(choice => {
        let btn = document.createElement("button");
        btn.innerText = choice;
        btn.onclick = () => checkAnswer(choice);
        choicesDiv.appendChild(btn);
    });

    time = 30;
    document.getElementById("time").innerText = time;
    clearInterval(timer);
    timer = setInterval(countdown, 1000);
}

function countdown() {
    time--;
    document.getElementById("time").innerText = time;
    if (time === 0) {
        nextQuestion();
    }
}

function checkAnswer(choice) {
    if (choice === questions[index].answer) {
        score++;
        document.getElementById("score").innerText = score;
    }
    nextQuestion();
}

function nextQuestion() {
    clearInterval(timer);
    index++;
    loadQuestion();
}

loadQuestion();
</script>
